//===== Ragnarok Offline mod: iRO Quests =====================
//= Alberta town quests (kRO's 2021 town quest revamp, on iRO)
//===== Description: =========================================
//= Kale was talked into exploring the shipwreck by a Zonda
//= promoter; his friend Timothy sends you after him. Mulberry,
//= the promoter, then wants the wreck to look like it holds
//= treasure, and the shipwreck researcher Aspar works out whose
//= ship it was: Drake's. His ghost asks for the wreck to be kept
//= clean. Knight Tybalt, meanwhile, sends you to Turtle Island
//= after the missing Templars. Every hunt becomes a daily, shared
//= with the Eden Group's mission boards (mod eden-rework-2021 uses
//= the same quest ids and cooldowns).
//=
//= Steps, NPCs and positions are the iRO client's (quest text
//= 21120-21161, navigation table); kill counts and daily rewards
//= are the Eden board's figures. Not in any source, chosen here:
//= story-step rewards (each pays like its daily), the suspicious
//= chest in the wreck, and the Knight Commander stands at the spot
//= the quest text gives (tur_dun03 108,74): rAthena's Knight
//= Leader of the older Turtle Island quest stands on iRO's navi
//= spot. Coins are hidden in the wreck's crates with mod
//= eden-rework-2021's character variables (eden_mb_coins,
//= eden_mb_crates) and its crates; this mod places the same five
//= crates only while that mod is off.
//=
//= Character variables: iq_alb21 (shipwreck step, 0-12),
//= iq_alb21t (Turtle Island step, 0-6).
//============================================================

	next;
	switch (select("Eating a Kukre liver (daily):Hiding old coins (daily):" + (iq_alb21 >= 12 ? "Cleaning the shipwreck 1F (daily):Cleaning the shipwreck 2F (daily)" : "") + ":Never mind")) {
	case 1: callfunc "F_IQ_TownDaily", 21127, 21126, 6500, 6500, 0, 0, "Mulberry", 40; end;
	case 2:
		if (checkquest(21129, PLAYTIME) == 2)
			erasequest 21129;
		mes "[Mulberry]";
		if (checkquest(21130) == 1) {
			if (eden_mb_coins < 10) {
				mes "10 coins, 2 per crate, first";
				mes "floor of the wreck.";

	case 5: close;
	}
	end;

S_Coins:
	setquest getarg(0);
	eden_mb_coins = 0;
	eden_mb_crates = 0;
	if (countitem(Antique_Coin) < 10)
		getitem Antique_Coin, 10 - countitem(Antique_Coin);
	return;

S_Refill:
	if (countitem(Antique_Coin) < 10 - eden_mb_coins) {
		next;
		mes "[Mulberry]";
		mes "Lost some? Here. Don't tell anyone.";
		getitem Antique_Coin, 10 - eden_mb_coins - countitem(Antique_Coin);
	}
	return;
}

// The wreck's crates, as in mod eden-rework-2021 (hidden while that mod is loaded).
-	script	Old Crate#iqalb	-1,{
	.@bit = 1 << atoi(charat(strnpcinfo(2), 5));
	if (checkquest(21128) != 1 && checkquest(21130) != 1) {
		mes "A battered old crate, half full of seawater.";
		close;
	}
	if (!eden_mb_coins)
		eden_mb_crates = 0;
	if (eden_mb_crates & .@bit) {
		mes "You've already hidden coins in this crate.";
		close;
	}
	if (eden_mb_coins >= 10) {
		mes "All 10 coins are hidden. Report to Mulberry.";
		close;
	}
	if (countitem(Antique_Coin) < 2) {
		mes "You need 2 Old Coins to hide here.";
		mes "Mulberry has more.";
		close;
	}
	mes "A battered old crate. Hide 2 Old Coins in it for treasure hunters to find?";
	next;
	if (select("Hide the coins:Leave it") == 2)
		close;
	progressbar "ffff00", 2;
	delitem Antique_Coin, 2;
	eden_mb_crates |= .@bit;
	eden_mb_coins += 2;
	if (eden_mb_coins >= 10)
		mes "That's the last of them. Report to Mulberry.";
	else
		mes "You tuck the coins under the rotting planks. " + (10 - eden_mb_coins) + " to go.";
	close;

OnInit:
	if (strnpcinfo(2) != "iqalb" && getnpcid(0, "Old Crate#edmb1"))
		disablenpc strnpcinfo(0);
	end;
}
