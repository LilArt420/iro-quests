//===== Ragnarok Offline mod: iRO Quests =====================
//= Morroc town quests (kRO's 2021 town quest revamp, on iRO)
//===== Description: =========================================
//= An old scholar studying the pyramid sends you after his lost
//= apprentice; the trail leads through the Thieves' Guild ruins,
//= Ernest's research team in the pyramid and the Sphinx. Each
//= step that is a hunt becomes a daily once done, shared with the
//= Eden Group's mission boards (mod eden-rework-2021 uses the same
//= quest ids and 04:00 cooldowns).
//=
//= Steps, NPCs and positions are the iRO client's (quest text
//= 21201-21220, navigation table); kill counts and daily rewards
//= are rAthena's quest_db and the Eden board's figures. Story-step
//= rewards are not in any source: each pays like its daily.
//=
//= Character variable: iq_moc21 (story step, 0-13).
//============================================================

morocc,151,86,4	script	Elderly Scholar#iqmoc1	897,{
	mes "[Elderly Scholar]";
	if (iq_moc21 == 0) {
		mes "Ah, a young adventurer. You look";
		mes "like you can walk a desert. I am";
		mes "studying the pyramid, and my";
		mes "apprentice... ran off.";
		next;
		mes "[Elderly Scholar]";
		mes "He went looking for the Assassins'";
		mes "Guild, in the desert southeast of";
		mes "Morroc. The fool! Would you find";
		mes "him and make sure he's alive?";
		next;
		if (BaseLevel < 40) {
			mes "[Elderly Scholar]";
			mes "...No, the desert would eat you.";
			mes "Come back when you're stronger.";
			mes "^777777(Base Level 40)^000000";
			close;
		}
		if (select("I'll find him.:Not now.") == 2)
			close;
		iq_moc21 = 1;
		setquest 21201;
		mes "[Elderly Scholar]";
		mes "Bless you. Southeast, in the";
		mes "Sograt Desert. Thin boy, too";
		mes "much courage.";
		close;
	}
	mes "My apprentice and I have moved";
	mes "our work to the oasis by the";
	mes "pyramid. Come find us there.";
	close;
}

moc_fild17,358,277,4	script	Wandering Boy#iqmoc	896,{
	if (iq_moc21 == 1) {
		mes "[Wandering Boy]";
		mes "The scholar sent you? I'm fine,";
		mes "I'm fine! I'm just... stuck.";
		mes "Sandmen everywhere, and they";
		mes "won't let me through.";
		next;
		mes "[Wandering Boy]";
		mes "Could you beat ^0000FF10 Sandmen^000000 and";
		mes "bring me ^0000FF5 Grit^000000 from them?";
		mes "I want to show the scholar what";
		mes "the desert's made of out here.";
		erasequest 21201;
		if (checkquest(21202) == 2)
			erasequest 21202;
		if (checkquest(21202) < 0)
			setquest 21202;
		iq_moc21 = 2;
		close;
	}
	if (iq_moc21 == 2) {
		mes "[Wandering Boy]";
		if (checkquest(21202, HUNTING) != 2 || countitem(Grit) < 5) {
			mes "10 Sandmen and 5 Grit, please!";
			close;
		}
		delitem Grit, 5;
		erasequest 21202;
		setquest 21203;
		getexp 10800, 9600;
		iq_moc21 = 3;
		mes "You did it! The path's clear.";
		mes "I'll go back to the scholar. He";
		mes "moved to the oasis in front of";
		mes "the pyramid. See you there!";
		close;
	}
	if (iq_moc21 >= 3) {
		callfunc "F_IQ_TownDaily", 21202, 21203, 10800, 9600, Grit, 5, "Wandering Boy", 50;
		end;
	}
	mes "[Wandering Boy]";
	mes "The Assassins' Guild is out here";
	mes "somewhere. I'm sure of it.";
	close;
}
