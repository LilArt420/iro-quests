//===== Ragnarok Offline mod: iRO Quests =====================
//= Aldebaran and Juno-field town quests (kRO's 2021 revamp, on iRO)
//===== Description: =========================================
//= A mother in Aldebaran sends her son's lunch up to the border
//= checkpoint, where Commander Cesare is short of hands and a Kiel
//= Hyre Academy student wants blanket materials. At the academy,
//= the guard, the Cat Paw Merchant and the student post their own
//= requests, and Ernst has a letter for his brother in Einbroch.
//= In Aldebaran, the Clock Tower Manager wants his tower cleaned,
//= from the books on 1F down to the witch underground. Every hunt
//= becomes a daily, shared with the Eden Group's mission boards
//= (mod eden-rework-2021 uses the same quest ids and cooldowns).
//=
//= Steps, NPCs and positions are the iRO client's (quest text
//= 21276-21293, 21300, navigation table); kill counts and daily
//= rewards are the Eden board's figures. Not in any source, chosen
//= here: the lunch run's and the blanket materials' rewards.
//=
//= Character variable: iq_ald21 (bits: 1 lunch delivered once,
//= 2 blanket materials delivered).
//============================================================

aldebaran,114,180,4	script	Middle-Aged Woman#iqald	74,{
	if (checkquest(21300, PLAYTIME) == 2)
		erasequest 21300;
	mes "[Middle-Aged Woman]";
	if (checkquest(21276) >= 0 && checkquest(21276) < 2) {
		mes "The checkpoint is north of town.";
		mes "My boy is the guard at the gate.";
		close;
	}
	if (checkquest(21300) >= 0) {
		mes "He ate it all? Good boy. I'll";
		mes "make another one tomorrow.";
		close;
	}
	mes "My son works at the ^0000FFborder";
	mes "checkpoint^000000 north of Aldebaran, and";
	mes "he never eats properly. Would you";
	mes "take him this lunch?";
	if (BaseLevel < 70) {
		mes "^777777(Base Level 70)^000000";
		close;
	}
	next;
	if (select("I'll take it:Not now") == 2)
		close;
	if (checkquest(21276) == 2)
		erasequest 21276;
	setquest 21276;
	mes "[Middle-Aged Woman]";
	mes "Thank you, dear. Careful, it's";
	mes "still warm.";
	close;
}

	mes "[Cerdin]";
	mes "My clock tower! Books turned into";

	next;
	switch (select("Cleaning the Clock Tower's First Floor (daily):The Living, Moving Clock 1 (daily):The Living, Moving Clock 2 (daily):Laughter Echoing from Underground (daily):Never mind")) {
	case 1: callfunc "F_IQ_TownDaily", 21286, 21287, 33820, 40650, 0, 0, "Cerdin", 70; end;
	case 2: callfunc "F_IQ_ClockDaily", 21288, 21289, 20310, 17370; end;
	case 3: callfunc "F_IQ_ClockDaily", 21290, 21291, 28320, 27960; end;
	case 4: callfunc "F_IQ_ClockDaily", 21292, 21293, 25720, 27280; end;
	case 5: close;
	}
}

// The clock dailies also pay a Key of Clock Tower and a Key of Underground (Eden board).
function	script	F_IQ_ClockDaily	{
	if (checkquest(getarg(0)) == 1 && checkquest(getarg(0), HUNTING) == 2)
		getitem Key_Of_Clock_Tower, 1;
	callfunc "F_IQ_TownDaily", getarg(0), getarg(1), getarg(2), getarg(3), 0, 0, "Cerdin", 80, Underground_Key, 1;
	end;
}
