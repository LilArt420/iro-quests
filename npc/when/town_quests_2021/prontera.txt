//===== Ragnarok Offline mod: iRO Quests =====================
//= Prontera town quests (kRO's 2021 town quest revamp, on iRO)
//===== Description: =========================================
//= Outside the west gate, Paul and Ann want the playground made
//= safe (Hornets, Rockers, Ambernite). In town, the Knights post a
//= notice for their regular Orc and Goblin subjugation: carry the
//= Knights' supplies to the Subjugation Leader in Orc Village and
//= his deputy at the Goblin settlement, and help with their hunts.
//= Priest Leader Lars has lost touch with Claude, sent to cleanse
//= the Glast Heim monastery, where Tatiana and Marylin help those
//= who fight; at the castle gate, Charon and Nadir hold back the
//= prisoners and demons. Every hunt becomes a daily, shared with
//= the Eden Group's mission boards (mod eden-rework-2021 uses the
//= same quest ids and cooldowns).
//=
//= Steps, NPCs and positions are the iRO client's (quest text
//= 21040-21045, 21246-21275, navigation table); kill counts and
//= daily rewards are the Eden board's figures. Not in any source,
//= chosen here: story-step rewards (each pays like its daily), the
//= supply run's reward, and what Tatiana's and Marylin's help is
//= (a heal, Blessing and Increase AGI, once every 5 minutes).
//=
//= Character variable: iq_prt21 (bits: 1 Hornets, 2 Rockers,
//= 4 Ambernite, 8 met the Leader, 16 met the Deputy, 32 found Claude).
//============================================================

prt_fild05,353,221,4	script	Paul#iqprt	706,{
	if (iq_prt21 & 1) {
		if (!(iq_prt21 & 2) && checkquest(21247) < 0) {
			mes "[Paul]";
			mes "You beat the bees! Then you're";
			mes "really strong. There's a grasshopper";
			mes "colony south of here, the ^0000FFRockers^000000.";
			mes "If they come, the playground's gone!";
			mes "Beat ^0000FF10 Rockers^000000, please?";
			setquest 21247;
			close;
		}
		if (checkquest(21247) >= 0 && checkquest(21247) < 2) {
			mes "[Paul]";
			if (checkquest(21247, HUNTING) != 2) {
				mes "10 Rockers, in the field south";
				mes "of here.";
				close;
			}
			erasequest 21247;
			getexp 1600, 1400;
			iq_prt21 = iq_prt21 | 2;
			mes "No more grasshoppers! You're the";
			mes "best. Ann's been scared of";
			mes "something too, go ask her.";
			close;
		}
		mes "[Paul]";
		mes "Want to keep the playground safe";
		mes "today too?";
		next;

		case 2: callfunc "F_IQ_TownDaily", 21250, 21253, 1600, 1400, 0, 0, "Paul", 10; end;
		case 3: close;
		}
	}
	mes "[Paul]";
	if (checkquest(21246) < 0) {
		mes "Hey! Are you strong? Mom says I";
		mes "can't play outside because a hive";
		mes "got knocked over and the bees went";
		mes "wild. Could you beat ^0000FF10 Hornets^000000?";

		mes "Yes! They're all over the field.";

		close;
	}

// Tatiana and Marylin help those on Claude's hunts.
function	script	F_IQ_GHPriest	{
	mes "[" + getarg(0) + "]";
	if (!(checkquest(21272) == 1 || checkquest(21274) == 1)) {
		mes "Lord, grant rest to the souls of";
		mes "this place...";
		close;
	}
	if (@iq_ghhelp > gettimetick(2)) {
		mes "Go, and God be with you. I'll pray";
		mes "for you again in a little while.";
		close;
	}
	mes "Claude sent you? Let me pray for";
	mes "you before you go on.";
	close2;
	@iq_ghhelp = gettimetick(2) + 300;
	percentheal 100, 100;
	specialeffect2 EF_HEAL2;
	sc_start SC_BLESSING, 240000, 10;
	specialeffect2 EF_BLESSING;
	sc_start SC_INCREASEAGI, 240000, 10;
	specialeffect2 EF_INCAGILITY;
	end;
}
