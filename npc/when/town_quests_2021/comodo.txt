//===== Ragnarok Offline mod: iRO Quests =====================
//= Comodo town quests (kRO's 2021 revamp, on iRO)
//===== Description: =========================================
//= Chef Bobby Q wants a new sauce from Que Sera, Sera, the herb
//= Serabi grows; his daughter Bunny Q puts it on her face
//= instead. Then Bobby Q's frogs, Bunny Q's tentacles and
//= Serabi's Megalodons become dailies. Comodo's Vigilantes and an
//= Explorer have their own hunts, with a one-time bonus for
//= turning the first one in in person. All dailies are shared
//= with the Eden Group's mission boards (mod eden-rework-2021
//= uses the same quest ids and cooldowns).
//=
//= Steps, NPCs and positions are the iRO client's (quest text
//= 21401-21421, 21442-21444, navigation table); kill counts and
//= daily rewards are the Eden board's. Not in any source, chosen
//= here: the story steps' rewards (like the dailies), the "frog
//= eggs" of the first step (Sticky Webfoots: there is no Frog Egg
//= item) and the one-time bonus (the hunt's EXP a second time).
//=
//= Character variables: iq_cmd21 (Que Sera, Sera step, 0-9),
//= iq_cmd21b (bits: 1, 2, 4 = bonus taken at each Vigilante or
//= the Explorer).
//============================================================

comodo,206,304,4	script	Chef Bobby Q#iqcmd	46,{
	mes "[Chef Bobby Q]";
	switch (iq_cmd21) {
	case 0:
		mes "A new sauce! The greatest sauce!";
		mes "I need ingredients: beat ^0000FF10 Thara";
		mes "Frogs^000000 in the Eastern Cave and bring";
		mes "me ^0000FF5 Sticky Webfoots^000000. Frog eggs,";
		mes "webfoots, all the same to a chef!";
		if (BaseLevel < 30) {
			mes "^777777(Base Level 30)^000000";
			close;
		}
		next;
		if (select("I'll get them:Not now") == 2)
			close;
		setquest 21401;
		iq_cmd21 = 1;
		close;
	case 1:
		if (checkquest(21401, HUNTING) != 2 || countitem(Sticky_Webfoot) < 5) {
			mes "10 Thara Frogs and 5 Sticky";
			mes "Webfoots, from the Eastern Cave!";
			close;
		}
		delitem Sticky_Webfoot, 5;
		erasequest 21401;
		getexp 5500, 5500;
		setquest 21402;
		iq_cmd21 = 2;
		mes "Wonderful. Now the secret: ^0000FFQue";
		mes "Sera, Sera^000000. Ask ^0000FFSerabi^000000 on the";
		mes "southern coast. She knows it best.";
		close;
	case 4:
		mes "The Que Sera, Sera? Sent to me?";
		mes "I never got it! Go to my house,";
		mes "east of here. Maybe it's there.";
		erasequest 21404;
		setquest 21405;
		iq_cmd21 = 5;
		close;
	case 8:
		if (countitem(Sticky_Webfoot) < 10) {
			mes "10 Sticky Webfoots, to my camp!";
			mes "From the Thara Frogs!";
			close;
		}
		delitem Sticky_Webfoot, 10;
		erasequest 21408;
		getexp 5500, 5500;
		iq_cmd21 = 9;
		mes "My daughter is all right, and I";
		mes "have my sauce. Que Sera, Sera!";
		mes "Come cook with me again any day.";
		close;
	}
	if (iq_cmd21 >= 9) {
		mes "Back for more cooking?";
		next;
		callfunc "F_IQ_TownDaily", 21411, 21410, 5500, 5500, Sticky_Webfoot, 10, "Chef Bobby Q", 30;
		end;
	}
	mes "The sauce! How is the sauce?";
	close;
}

// The Vigilantes' and the Explorer's hunts: the first one turned in in person
// pays a one-time bonus and records 21442-21444.
// getarg: 0 daily, 1 cooldown, 2 bonus record, 3 bit, 4 Base EXP, 5 Job EXP, 6 NPC name
function	script	F_IQ_CmdHunt	{
	if (!(iq_cmd21b & getarg(3)) && checkquest(getarg(0)) == 1 && checkquest(getarg(0), HUNTING) == 2) {
		iq_cmd21b = iq_cmd21b | getarg(3);
		setquest getarg(2);
		completequest getarg(2);
		getexp getarg(4), getarg(5);
	}
	callfunc "F_IQ_TownDaily", getarg(0), getarg(1), getarg(4), getarg(5), 0, 0, getarg(6), 60;
	end;
}
