//===== Ragnarok Offline mod: iRO Quests =====================
//= Payon town quests (kRO's 2021 town quest revamp, on iRO)
//===== Description: =========================================
//= An old man's well and a missing grandson lead into Payon
//= Cave with Reno; the Archer Village guard, the elder and the
//= fox spirit of the abandoned village follow. Beside it, Payon's
//= neighbours have small requests (Dong Dong's firewood, a boa,
//= an "onion", mushrooms), and the Wandering Monk and the young
//= apprentice post dailies in the cave. Every hunt turns into a
//= daily once done, shared with the Eden Group's mission boards
//= (mod eden-rework-2021 uses the same quest ids and cooldowns).
//=
//= Steps, NPCs and positions are the iRO client's (quest text
//= 21011-21046, navigation table); kill counts and daily rewards
//= are the Eden board's figures. Not in any source, chosen here:
//= story-step rewards (each pays like its daily), the fox hunt's
//= reward, and the Archer Village guard stands 2 cells south of
//= iRO's spot (a Warper is there on this server). iRO's reunion
//= scene in Archer Village (a second grandfather, Reno and guards)
//= is left out: rAthena's own NPCs stand on those cells.
//=
//= Character variables: iq_pay21 (main story, 0-8), iq_pay_dd
//= (Dong Dong, 0-4), iq_pay_side (bits: 1 lady, 2 grandma, 4 grandpa).
//============================================================

payon,192,109,4	script	Village Grandfather#iqpay	842,{
	mes "[Village Grandfather]";
	if (checkquest(21019) >= 0 && checkquest(21019) < 2 && countitem(Liqueur_Of_Lotus) < 10 && countitem(FaintBreathing) < 10) {
		mes "The monk wants to calm the poor";
		mes "souls with a drink? Here, ^0000FFHa";
		mes "Soju^000000. The good stuff.";
		getitem Liqueur_Of_Lotus, 10 - countitem(Liqueur_Of_Lotus);
		close;
	}
	switch (iq_pay21) {
	case 0:
		mes "Young one! My back won't let me";
		mes "bend over the well, and my liquor";
		mes "is chilling in a stave down there.";
		mes "Durevak's own recipe. Would you";
		mes "pull it up for me?";
		next;
		if (select("Sure:Not now") == 2)
			close;
		iq_pay21 = 1;
		setquest 21011;
		mes "[Village Grandfather]";
		mes "The well right beside me. Pull";
		mes "slowly, the rope is old.";
		close;
	case 1:
		mes "The well right beside me!";
		close;
	case 2:
		mes "Ahh, still cold. Thank you!";
		getexp 800, 800;
		next;
		mes "[Village Grandfather]";
		mes "Now... I hate to ask more, but my";
		mes "grandson snuck off to the Archer";
		mes "Village this morning and hasn't";
		mes "come back. The ^0000FFguard by the cave^000000";
		mes "might have seen him.";
		erasequest 21011;
		setquest 21012;
		iq_pay21 = 3;
		close;
	case 3:
	case 4:
	case 5:
		mes "Any news of my grandson? The guard";
		mes "by the cave in Archer Village...";
		close;
	default:
		mes "My grandson is home, sulking.";
		mes "Good! Thank you, young one.";
		close;
	}
}
