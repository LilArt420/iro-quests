//===== Ragnarok Offline mod: iRO Quests =====================
//= Izlude town quests (kRO's 2021 town quest revamp, on iRO)
//===== Description: =========================================
//= Instructor Pontes trains new adventurers on the locals'
//= complaints (Peco Pecos and Mukas, Magnolias and Metallers),
//= then sends them to Byalan Island: Student Helden needs
//= specimens and the Wandering Store Owner wants claws. In the
//= undersea tunnel, the injured Ernest worries about Leader
//= Maxim's team, who are cooking, drinking and chasing "Poseidon"
//= on the floor below. Every hunt becomes a daily, shared with the
//= Eden Group's mission boards (mod eden-rework-2021 uses the same
//= quest ids and cooldowns).
//=
//= Steps, NPCs and positions are the iRO client's (quest text
//= 21221-21244, navigation table); kill counts and daily rewards
//= are the Eden board's figures. Not in any source, chosen here:
//= story-step rewards (each pays like its daily, or the nearest
//= one) and the Store Owner's first-claw reward.
//=
//= Character variable: iq_izl21 (bits: 1 Pontes 1, 2 Pontes 2,
//= 4 met Helden, 8 Store Owner, 16 met Maxim, 32 Kusekili,
//= 64 Ankana, 128 soup delivered, 256 Maxim's proof).
//============================================================

izlude,132,207,4	script	Instructor Pontes#iqizl	745,{
	mes "[Instructor Pontes]";
	if (!(iq_izl21 & 1)) {
		if (checkquest(21221) < 0) {
			mes "A true adventurer helps people.";
			mes "Let's get you used to requests.";
			mes "The Sograt Desert folk complain";
			mes "about ^0000FFPeco Pecos and Mukas^000000. Beat";
			mes "^0000FF10 of each^000000 for practice.";
			if (BaseLevel < 20) {
				mes "^777777(Base Level 20)^000000";
				close;
			}
			next;
			if (select("I'll do it:Not now") == 2)
				close;
			setquest 21221;
			close;
		}
		if (checkquest(21221, HUNTING) != 2) {
			mes "10 Peco Pecos and 10 Mukas, in";
			mes "the Sograt Desert.";
			close;
		}
		erasequest 21221;
		getexp 4800, 4200;
		iq_izl21 = iq_izl21 | 1;
		mes "Well done! Helping each other,";
		mes "that's what makes an adventurer.";
		mes "Come to me for more requests.";
		close;
	}
	if (!(iq_izl21 & 2) && BaseLevel >= 40) {
		if (checkquest(21224) < 0) {
			mes "Ready for a harder one? South of";
			mes "Izlude, ^0000FFMagnolias and Metallers^000000";
			mes "bother the travelers. Eggs and";
			mes "grasshoppers! ^0000FF10 of each^000000.";
			setquest 21224;
			close;
		}
		if (checkquest(21224, HUNTING) != 2) {
			mes "10 Magnolias and 10 Metallers,";
			mes "in the field south of Izlude.";
			close;
		}
		erasequest 21224;
		getexp 8400, 8380;
		iq_izl21 = iq_izl21 | 2;
		mes "Excellent. You're ready for real";
		mes "work. A university student on";
		mes "^0000FFByalan Island^000000 asked for help.";
		mes "Take the boat from the harbor.";
		setquest 21227;
		close;
	}
	mes "Which request will you take today?";
	next;
	switch (select("The Noisy Ones (daily):" + ((iq_izl21 & 2) ? "Eggs and Grasshoppers (daily)" : "") + ":Never mind")) {
	case 1: callfunc "F_IQ_TownDaily", 21222, 21223, 4800, 4200, 0, 0, "Instructor Pontes", 20; end;
	case 2: callfunc "F_IQ_TownDaily", 21225, 21226, 8400, 8380, 0, 0, "Instructor Pontes", 40; end;
	case 3: close;
	}
}

S_Soup:
	if ((iq_izl21 & 96) == 96 && checkquest(21236) < 0 && !(iq_izl21 & 128)) {
		next;
		mes "[Leader Maxim]";
		mes "Take a bowl of soup to Ernest, on";
		mes "the floor above. Tell him \"We're";
		mes "fine!\". That's all he needs.";
		setquest 21236;
	}
	return;
}
