	mes "[Leader Maxim]";
	if (checkquest(21350) < 0) {
		mes "Warn us? About Juperos? Ha! We're";
		mes "researchers, we know the dangers.";
		mes "...Mostly. Well, since you're here,";
		mes "you could help.";
		setquest 21350;
		completequest 21350;
		next;
		mes "[Leader Maxim]";
	}
	mes "The Dimiks come in every color. For";
	mes "my research, I need them... fewer.";
	next;
	switch (select("Colorful Dimiks - 1 (daily):Colorful Dimiks - 2 (daily):Never mind")) {
	case 1: callfunc "F_IQ_TownDaily", 21355, 21356, 300000, 250000, 0, 0, "Leader Maxim", 110, 25223, 1; end;
	case 2: callfunc "F_IQ_TownDaily", 21357, 21358, 300000, 250000, 0, 0, "Leader Maxim", 110, 25223, 1; end;
	case 3: close;
	}
}
