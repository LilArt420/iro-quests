//===== Ragnarok Offline mod: iRO Quests =====================
//= Einbroch, Einbech and Lighthalzen town quests (kRO's 2021
//= town quest revamp, on iRO)
//===== Description: =========================================
//= Ernst at the Kiel Hyre Academy has a letter for his brother
//= Friedrich in the factory district; Friedrich's commuter pass
//= takes you to Einbech, where Guard Artin needs help in the
//= mine. In Einbroch, Guard Eisen fights the toxic gas and the
//= Teddy Bears and keeps forgetting to remind Karl of his shift;
//= in Lighthalzen, Theodore takes everyone's complaints and Reno
//= is afraid of ghosts. Every hunt becomes a daily, shared with
//= the Eden Group's mission boards (mod eden-rework-2021 uses the
//= same quest ids and cooldowns).
//=
//= Steps, NPCs and positions are the iRO client's (quest text
//= 21301-21324, navigation table); kill counts and daily rewards
//= are the Eden board's figures. Not in any source, chosen here:
//= the letter's and the shift notice's rewards.
//=
//= Character variable: iq_ein21 (bits: 1 letter delivered,
//= 2 commuter pass used, 4 heard about Einbech, 8 Karl told).
//============================================================

yuno_fild08,172,194,4	script	Ernst Schumacher#iqein	904,{
	mes "[Ernst Schumacher]";
	if (iq_ein21 & 1) {
		mes "My brother wrote back! Thank you.";
		close;
	}
	if (checkquest(21301) >= 0) {
		mes "My brother Friedrich works in the";
		mes "factory district west of the Abyss";
		mes "Valley, near Einbroch.";
		close;
	}
	mes "I'm a student here. Would you take";
	mes "a letter to my older brother,";
	mes "^0000FFFriedrich^000000? He works in the factory";
	mes "district outside Einbroch.";
	if (BaseLevel < 80) {
		mes "^777777(Base Level 80)^000000";
		close;
	}
	next;
	if (select("I'll take it:Not now") == 2)
		close;
	setquest 21301;
	mes "[Ernst Schumacher]";
	mes "Tell him I'm eating well. Even if";
	mes "I'm not.";
	close;
}

ein_fild06,163,173,4	script	Friedrich Schumacher#iqein	851,{
	mes "[Friedrich Schumacher]";
	if (checkquest(21301) >= 0 && checkquest(21301) < 2) {
		mes "A letter from Ernst? \"Eating";
		mes "well.\" Liar. ...Thank you.";
		erasequest 21301;
		getexp 10000, 10000;
		iq_ein21 = iq_ein21 | 1;
		next;
		mes "[Friedrich Schumacher]";
		mes "Say, Metalings and Holden dig";
		mes "around the factory and chew our";
		mes "equipment. If you clean them up,";
		mes "I'll lend you my ^0000FFcommuter pass^000000.";
		mes "It goes straight to Einbech.";
		close;
	}
	if (!(iq_ein21 & 1)) {
		mes "Factory work. Dust, noise, and";
		mes "Metalings.";
		close;
	}
	switch (select("Please Help Clean Up the Area (daily):Use the commuter pass to Einbech:Never mind")) {
	case 1:
		callfunc "F_IQ_TownDaily", 21302, 21303, 40920, 38220, 0, 0, "Friedrich Schumacher", 80;
		end;
	case 2:
		mes "[Friedrich Schumacher]";
		mes "Here. Show it to Guard Artin when";
		mes "you get off.";
		close2;
		if (!(iq_ein21 & 2) && checkquest(21304) < 0)
			setquest 21304;
		warp "einbech", 154, 196;
		end;
	case 3:
		close;
	}
}
