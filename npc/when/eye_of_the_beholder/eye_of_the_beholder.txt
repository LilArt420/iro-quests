//===== Ragnarok Offline mod: iRO Quests =====================
//= Eye of the Beholder (iRO)
//===== Description: =========================================
//= Beho, a beholder living in Rachel Temple, has fallen for
//= Bianca, who is terrified of beholders. Help him win her
//= over: chocolate from Kyara, a song with Doina, a necklace
//= from Lilla, a bouquet from Kees and the Pope's blessing.
//= Base level 50. Reward: 500,000 Base EXP, 375,000 Job EXP,
//= 5 Yggdrasil Berries and 1 Eden Merit Badge (iRO's non-VIP
//= reward).
//=
//= Steps, positions and rewards follow iROWiki's "Eye of the
//= Beholder"; quest-log ids are the iRO client's own
//= (61001-61029). The dialogue is written for this mod.
//=
//= Character variable: iq_beho = the quest-log entry the
//= character is on (0 = not started, 1 = done).
//============================================================

// Starts the quest, and takes anyone to Beho.
airplane_01,238,57,4	script	Beholder#iqbeho1	1633,{
	mes "[Beholder]";
	if (iq_beho == 0) {
		mes "Psst. Over here, in the shadow.";
		mes "Don't scream. Everyone screams.";
		mes "I'm not like the ones that float";
		mes "around Juperos zapping people.";
		next;
		mes "[Beholder]";
		mes "My friend ^0000FFBeho^000000 lives in Rachel";
		mes "Temple. He's in trouble... the";
		mes "heart kind of trouble. Could you";
		mes "talk to him? I can send you there.";
		next;
		if (BaseLevel < 50) {
			mes "[Beholder]";
			mes "...Hm, on second thought, the";
			mes "temple is no place for a beginner.";
			mes "Come back when you're stronger.";
			mes "^777777(Base Level 50)^000000";
			close;
		}
		if (select("Sure, send me to Beho.:No, thanks.") == 2) {
			mes "[Beholder]";
			mes "Everyone says that. *sigh*";
			close;
		}
		iq_beho = 61001;
		setquest 61001;
		mes "[Beholder]";
		mes "Thank you! Hold still...";
		close2;
		warp "ra_temin", 268, 318;
		end;
	}
	if (iq_beho == 1) {
		mes "Beho and Bianca send their love.";
		mes "Literally. They won't stop.";
		close;
	}
	mes "Want to go back to Beho?";
	next;
	if (select("Yes, please.:Not now.") == 2)
		close;
	close2;
	warp "ra_temin", 268, 318;
	end;
}

				mes "...That's not how a love song";

		mes "say hello can't be scary.";
