//===== Ragnarok Offline mod: iRO Quests =====================
//= Overlook Water Dungeon (iRO)
//===== Description: =========================================
//= iRO's copy of the Byalan dungeon, haunted by tougher "Aged"
//= monsters. The Sign, west of Geffen, lets adventurers of Base
//= Level 30-85 in. A Sign Board on each of the 5 floors posts
//= one hunt (not repeatable); after all 5, any board lets you
//= pick one Byalan card.
//=
//= Hunts, rewards and positions follow iROWiki's "Overlook Water
//= Dungeon Quests"; quest-log ids are the iRO client's own
//= (50531-50535). On iRO the dungeon is for VIP (or two Magical
//= Stones for mini-VIP); this server has no VIP, so it is open.
//= The EXP is paid in two halves, as on iRO.
//=
//= Character variables: iq_owd (hunts finished, 0-5),
//= iq_owd_card (1 = the final card was chosen).
//============================================================

gef_fild07,264,286,4	script	The Sign#iqowd	837,{
	mes "[The Sign]";
	mes "^0000FFOverlook Water Dungeon^000000";
	mes "Below lies a flooded twin of the";
	mes "Byalan caves, where the sea";
	mes "creatures have grown old and";
	mes "strong. Only adventurers of";
	mes "^0000FFBase Level 30 ~ 85^000000 may enter.";
	next;
	if (BaseLevel < 30 || BaseLevel > 85) {
		mes "[The Sign]";
		mes "^777777The water below does not";
		mes "answer you. (Base Level 30 ~ 85)^000000";
		close;
	}
	if (select("Enter the dungeon:Leave") == 2)
		close;
	close2;
	warp "iz_dun00t", 168, 167;
	end;
}

// One board per floor; floor = the digit in the hidden name.
iz_dun00t,174,169,4	script	Sign Board#iqowd1	837,{
	.@f = atoi(charat(strnpcinfo(2), 5));		// 1..5
	setarray .@q[1], 50531, 50532, 50533, 50534, 50535;
	setarray .@m1$[1], "Aged Plankton", "Aged Vadon", "Aged Obeaune", "Aged Marc", "Aged Strouf";
	setarray .@m2$[1], "Aged Marina", "Aged Kukre", "Aged Marse", "Aged Phen", "Aged Merman";
	setarray .@n[1], 20, 40, 60, 80, 100;
	setarray .@be[1], 60660, 88695, 155700, 218475, 335340;
	setarray .@je[1], 68040, 99900, 174960, 245925, 377190;
	.@qid = .@q[.@f];
	mes "[Sign Board]";
	if (BaseLevel < 30 || BaseLevel > 85) {
		mes "^777777The notice is too waterlogged to";
		mes "read. (Base Level 30 ~ 85)^000000";
		close;
	}
	if (iq_owd >= 5) {
		if (iq_owd_card) {
			mes "Every notice on this board is";
			mes "marked ^0000FFDONE^000000 in your hand.";
			mes "The Overlook has no more work";
			mes "for you.";
			close;
		}
		mes "You have cleared all five floors.";
		mes "Choose a reward from the old";
		mes "Byalan collection, carefully:";
		mes "^FF0000you can only choose once.^000000";
		next;
		setarray .@card[0], Kukre_Card, Vadon_Card, Plankton_Card, Hydra_Card, Marina_Card, Cornutus_Card, Marse_Card, Obeaune_Card, Merman_Card, Marc_Card, Phen_Card, Sword_Fish_Card, Marine_Sphere_Card, Strouf_Card, Deviace_Card;
		.@menu$ = "";
		for (.@i = 0; .@i < getarraysize(.@card); .@i++)
			.@menu$ += getitemname(.@card[.@i]) + ":";
		.@c = select(.@menu$ + "Not now") - 1;
		if (.@c == getarraysize(.@card))
			close;
		mes "[Sign Board]";
		mes "^0000FF" + getitemname(.@card[.@c]) + "^000000?";
		next;
		if (select("Yes, this one.:Let me think again.") == 2)
			close;
		iq_owd_card = 1;
		getitem .@card[.@c], 1;
		mes "[Sign Board]";
		mes "A card slips out from behind the";
		mes "notices. The Overlook thanks you.";
		close;
	}
	if (iq_owd >= .@f) {
		mes "^0000FFDONE^000000 is written across this";
		mes "floor's notice.";
		close;
	}
	if (iq_owd < .@f - 1) {
		mes "The notice here is for those who";
		mes "have cleared the floors above.";
		mes "Start with the Sign Board on";
		mes "floor " + (iq_owd + 1) + ".";
		close;
	}
	mes "^0000FFWANTED^000000: ^FF0000" + .@n[.@f] + "^000000 " + .@m1$[.@f] + " and";
	mes "^FF0000" + .@n[.@f] + "^000000 " + .@m2$[.@f] + ".";
	mes "Reward: " + callfunc("F_InsertComma", .@be[.@f]) + " Base EXP,";
	mes callfunc("F_InsertComma", .@je[.@f]) + " Job EXP and supplies.";
	next;
	if (checkquest(.@qid) < 0) {
		if (select("Take the notice:Leave it") == 2)
			close;
		setquest .@qid;
		mes "[Sign Board]";
		mes "You take the notice. Your quest";
		mes "log will keep count.";
		close;
	}
	if (checkquest(.@qid, HUNTING) != 2) {
		mes "[Sign Board]";
		mes "Not done yet. Check your quest";
		mes "log (Alt+U).";
		close;
	}
	if (.@f == 1)
		.@c = select("Guard [1]:Buckler [1]");
	else if (.@f == 2)
		.@c = select("Muffler [1]:Manteau [1]");
	mes "[Sign Board]";
	mes "The board creaks, and a reward";
	mes "floats up from the water.";
	completequest .@qid;
	iq_owd = .@f;
	getexp .@be[.@f] / 2, .@je[.@f] / 2;
	getexp .@be[.@f] - .@be[.@f] / 2, .@je[.@f] - .@je[.@f] / 2;
	switch (.@f) {
	case 1: getitem (.@c == 1 ? Guard_ : Buckler_), 1; break;
	case 2: getitem (.@c == 1 ? Muffler_ : Manteau_), 1; break;
	case 3: getitem Elunium, 2; break;
	case 4: getitem Rosary, 1; break;
	case 5: getitem Rosary, 1; getitem Elunium, 2; break;
	}
	if (.@f == 5) {
		next;
		mes "[Sign Board]";
		mes "All five floors are clear! Talk";
		mes "to any Sign Board to choose your";
		mes "final reward.";
	}
	close;

OnInit:
	.@f = atoi(charat(strnpcinfo(2), 5));
	setarray .@q[1], 50531, 50532, 50533, 50534, 50535;
	questinfo QTYPE_QUEST2, QMARK_YELLOW, "checkquest(" + .@q[.@f] + ",HUNTING) == 2";
	end;
}
